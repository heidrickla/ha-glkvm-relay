#include <tunables/global>

profile glkvm_relay flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>
  #include <abstractions/openssl>
  #include <abstractions/ssl_certs>

  # The web console listens on 443 on the host network.
  capability net_bind_service,

  # Files: the rule set the Home Assistant example profiles use for an
  # s6-overlay based image, so init and the wrapper script can run.
  file,
  signal (send) set=(kill,term,int,hup,cont),
  signal (receive) set=(kill,term,int,hup,cont),

  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,
  /tmp/** rwk,
  /run.sh ix,

  # options, the relay's database, TURN credentials, configs and the self-signed certificate
  /data/** rwk,
  # the certificate pair from Home Assistant's /ssl
  /ssl/** r,
  # rttys reads its certificate and database through links under /home
  /home/** rwkl,
  /usr/share/glkvm/** r,

  # web console, device and proxy ports, the Supervisor API; STUN and TURN over UDP;
  # `ip route get` asks the kernel through netlink
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network netlink raw,
}
